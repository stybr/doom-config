(defun my/latex-matrix (&optional type opt)
  "Zeptá se na rozměr a vloží matici s poli na vyplnění.
OPT jsou volby za \\begin, např. \"[r]\"."
  (interactive)
  (let* ((type (or type "pmatrix"))
         (dims (mapcar #'string-to-number
                       (split-string (read-string "Rozměr (např. 3x4): ") "[x ]+" t)))
         (rows (nth 0 dims))
         (cols (or (nth 1 dims) rows))
         (n 0))
    (yas-expand-snippet
     (concat "\\begin{" type "}" (or opt "") "\n"
             (mapconcat
              (lambda (_r)
                (mapconcat (lambda (_c) (format "$%d" (setq n (1+ n))))
                           (number-sequence 1 cols) " & "))
              (number-sequence 1 rows) " \\\\\\\\\n")
             "\n\\end{" type "}$0"))))
#+end_src

** Horner generator
#+begin_src emacs-lisp
(defun my/num->string (n)
  (if (integerp n) (number-to-string n) (format "%g" n)))

(defun my/horner (coeffs root)
  "Vloží Hornerovo schéma pro COEFFS a kořen ROOT.
Dělení se opakuje, dokud vychází nulový zbytek."
  (interactive
   (list (mapcar #'string-to-number
                 (split-string (read-string "Koeficienty (od nejvyšší mocniny): ")))
         (read-number "Kořen: ")))
  (let* ((n (length coeffs))
         (cur coeffs)
         (lines (list (concat "  & "
                              (mapconcat #'my/num->string coeffs " & ")
                              " \\\\"))))
    (catch 'done
      (while t
        (let* ((acc (car cur)) (out (list acc)) (prods nil))
          (dolist (c (cdr cur))
            (let ((p (* acc root)))
              (setq acc (+ c p))
              (push p prods)
              (push acc out)))
          (setq out (nreverse out) prods (nreverse prods))
          (push (concat (my/num->string root) " &   & "
                        (mapconcat #'my/num->string prods " & ")
                        " \\\\")
                lines)
          (push "\\hline" lines)
          (push (concat "  & " (mapconcat #'my/num->string out " & ") " \\\\")
                lines)
          (if (and (zerop (car (last out))) (> (length out) 2))
              (setq cur (butlast out))
            (throw 'done nil)))))
    (insert (format "\\[\n\\begin{array}{c|%s}\n%s\n\\end{array}\n\\]"
                    (make-string n ?c)
                    (string-join (nreverse lines) "\n")))))
#+end_src

#+begin_src emacs-lisp
(defun my/quadratic (a b c)
  "Vloží výpočet kořenů kvadratické rovnice, včetně komplexních."
  (interactive "nKoeficient a: \nnKoeficient b: \nnKoeficient c: ")
  (let* ((d (- (* b b) (* 4 a c)))
         (ds (my/num->string d))
         (as (my/num->string a))
         (re (/ (- b) (* 2.0 a))))
    (insert
     (format "\\begin{align*}
%s &= 0 \\\\
D &= b^2 - 4ac = (%s)^2 - 4 \\cdot %s \\cdot %s = %s \\\\
"
             (my/polynom-string (list a b c))
             (my/num->string b) as (my/num->string c) ds))
    (cond
     ((< d 0)
      (let ((im (/ (sqrt (- d)) (* 2.0 a))))
        (insert (format "x_{1,2} &= \\frac{-b \\pm i\\sqrt{|D|}}{2a} = \\frac{%s \\pm i\\sqrt{%s}}{%s} \\\\
x_1 &= %s + %si, \\quad x_2 = %s - %si
\\end{align*}\n"
                        (my/num->string (- b)) (my/num->string (- d))
                        (my/num->string (* 2 a))
                        (my/num->string re) (my/num->string im)
                        (my/num->string re) (my/num->string im)))))
     ((= d 0)
      (insert (format "x &= \\frac{-b}{2a} = %s
\\end{align*}\n"
                      (my/num->string re))))
     (t
      (let ((s (sqrt d)))
        (insert (format "x_{1,2} &= \\frac{-b \\pm \\sqrt{D}}{2a} = \\frac{%s \\pm \\sqrt{%s}}{%s} \\\\
x_1 &= %s, \\quad x_2 = %s
\\end{align*}\n"
                        (my/num->string (- b)) ds (my/num->string (* 2 a))
                        (my/num->string (/ (+ (- b) s) (* 2.0 a)))
                        (my/num->string (/ (- (- b) s) (* 2.0 a))))))))))
#+end_src

(defun my/polynom-string (coeffs &optional var)
  "Vrátí LaTeX zápis polynomu z COEFFS (od nejvyšší mocniny)."
  (let* ((var (or var "x"))
         (n (1- (length coeffs)))
         (parts nil))
    (dolist (c coeffs)
      (let* ((abs-c (abs c))
             (znak (if (< c 0) "-" "+"))
             (mocnina (cond ((= n 0) "")
                            ((= n 1) var)
                            (t (format "%s^{%d}" var n))))
             (cislo (if (and (= abs-c 1) (> n 0)) "" (my/num->string abs-c))))
        (unless (zerop c)
          (push (format "%s %s%s" znak cislo mocnina) parts)))
      (setq n (1- n)))
    (let ((s (string-join (nreverse parts) " ")))
      (if (string-prefix-p "+ " s) (substring s 2) s))))

(defun my/polynom (coeffs)
